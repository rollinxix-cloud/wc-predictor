    <script>
        async function loadCachedWorldCupData() {
            try {
                const response = await fetch('worldcup.json?t=' + new Date().getTime());
                const data = await response.json();
                renderMatches(data.matches);
            } catch (error) {
                console.error("Local snapshot read error:", error);
                document.getElementById('live-matches-grid').innerHTML = `
                    <div class="col-span-2 border border-slate-800 bg-slate-900/50 text-slate-400 p-6 rounded-xl text-xs text-center italic">
                        Initial database sync in progress. The cloud engine will publish live fixtures profiles shortly!
                    </div>`;
            }
        }

        function renderMatches(matches) {
            const grid = document.getElementById('live-matches-grid');
            const groupsContainer = document.getElementById('groups-container');
            grid.innerHTML = '';
            groupsContainer.innerHTML = '';

            const liveMatches = matches.filter(m => ['IN_PLAY', 'PAUSED'].includes(m.status));
            
            if(liveMatches.length === 0) {
                grid.innerHTML = `<div class="col-span-2 text-center py-8 text-slate-500 italic text-xs border border-slate-900 bg-slate-950/20 rounded-2xl">No games are actively live on the pitches right now. Standby for next match block kicks!</div>`;
            } else {
                liveMatches.forEach(match => {
                    grid.insertAdjacentHTML('beforeend', createMatchCard(match, true));
                });
            }

            const stages = {};
            matches.forEach(match => {
                const stageName = match.group || match.stage;
                if (!stages[stageName]) stages[stageName] = [];
                stages[stageName].push(match);
            });

            Object.keys(stages).sort().forEach(stage => {
                let stageHtml = `
                    <div class="bg-slate-900/40 border border-slate-800/80 rounded-2xl p-4 space-y-3 shadow-xl">
                        <h3 class="text-xs font-black tracking-wider text-emerald-400 uppercase border-b border-slate-800 pb-2">
                            ${stage.replace('_', ' ')}
                        </h3>
                        <div class="space-y-2 divide-y divide-slate-800/40">`;

                stages[stage].slice(0, 4).forEach(match => {
                    const homeScore = match.score.fullTime.home !== null ? match.score.fullTime.home : '-';
                    const awayScore = match.score.fullTime.away !== null ? match.score.fullTime.away : '-';
                    stageHtml += `
                        <div class="flex justify-between items-center pt-2 text-[11px]">
                            <span class="text-slate-300 font-medium">${match.homeTeam.name}</span>
                            <span class="font-mono bg-slate-950 px-2 py-0.5 rounded border border-slate-850 font-bold text-amber-400">${homeScore} : ${awayScore}</span>
                            <span class="text-slate-300 font-medium text-right">${match.awayTeam.name}</span>
                        </div>`;
                });

                stageHtml += `</div></div>`;
                groupsContainer.insertAdjacentHTML('beforeend', stageHtml);
            });
        }

        function createMatchCard(match, isLive) {
            const homeScore = match.score.fullTime.home !== null ? match.score.fullTime.home : 0;
            const awayScore = match.score.fullTime.away !== null ? match.score.fullTime.away : 0;
            return `
                <div class="bg-gradient-to-br from-slate-900 to-slate-950 border ${isLive ? 'border-emerald-500/40 shadow-[0_0_15px_rgba(16,185,129,0.1)]' : 'border-slate-800'} p-5 rounded-2xl">
                    <div class="flex justify-between items-center mb-3">
                        <span class="text-[9px] font-black uppercase tracking-widest text-slate-500">${match.group || match.stage}</span>
                        <span class="bg-red-500/10 text-red-400 border border-red-500/30 text-[9px] font-black px-2 py-0.5 rounded uppercase tracking-wider animate-pulse">● LIVE MATCH</span>
                    </div>
                    <div class="grid grid-cols-3 items-center text-center">
                        <span class="font-bold text-xs text-slate-200">${match.homeTeam.name}</span>
                        <div class="flex items-center justify-center gap-3">
                            <span class="text-xl font-mono font-black text-emerald-400">${homeScore}</span>
                            <span class="text-[10px] font-bold text-slate-600">VS</span>
                            <span class="text-xl font-mono font-black text-emerald-400">${awayScore}</span>
                        </div>
                        <span class="font-bold text-xs text-slate-200">${match.awayTeam.name}</span>
                    </div>
                </div>`;
        }

        loadCachedWorldCupData();
    </script>
